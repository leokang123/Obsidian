```dataviewjs
const list = dv.pages('#일반').filter(p => p.file.folder === "Studied");  

// 태그에 하나라도 있으면 거름: filterItem1
const filterItem1 = ["대학교", "회고"]; 
// 해당 주제 거름: filterItem2 
const filterItem2 = ["일반", "중요", "잡", "학습정리", "네이버", "개념", "프로젝트", "책리뷰"];  
  
const filterPage = list.filter(p => !p.tags.some(q => filterItem1.includes(q)));  
const sortPage = filterPage.sort(page => page.updated, "desc");  
  
// 주제 필드만 모으기  
const subjectSet = new Set(  
sortPage  
.map(p => p.주제)  
.filter(v => v) // 주제가 없는 페이지 제외  
);  
  
const subjectList = [...subjectSet];  
// const filterList = subjectList.filter(item => !filterItem2.includes(item));  
  
let res = '';  
subjectList.forEach((subject) => {  
res += `~~~ dataview  
TABLE WITHOUT ID  
link(file.path,file.name) AS "${subject}"  
FROM "Studied"  
WHERE 주제 = "${subject}" AND !contains(file.tags, "대학교")  
SORT series DESC  
~~~\n<br>\n\r`;  
});  
  
dv.paragraph(res);
```
