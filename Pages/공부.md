````ad-cyan
const list = dv.pages('#일반').filter(p => p.file.folder === "Studied");
// const semesters = ["3-1","3-2","4-1","4-2"]
const filterItem1 = ["대학교", "회고"] //개인 학습과 거리가 먼거 
const filterItem2 = ["일반", "중요", "잡", "학습정리", "네이버", "개념", "프로젝트", "책리뷰"]; // 세부적인 주제에 관한것만 
const filterPage = list.filter(p => !p.tags.some(q => filterItem1.includes(q))); // 학습과 관련된거 필터링
const sortPage = filterPage.sort(page => page.updated, "desc")
const tagSet = new Set(sortPage.tags);
const tagList = [...tagSet];
const filterList = tagList.filter((item) => !filterItem2.includes(item)) // 세부적인 학습에 관련된거 필터링
let res = '';
filterList.forEach((tag) => {
res += `~~~ dataview
TABLE WITHOUT ID
link(file.path,file.name) AS "${tag}"
FROM "Studied" and #${tag}
WHERE !contains(file.tags, "대학교")
SORT 주제, series DESC
~~~\n<br>\n\r`;
})
dv.paragraph(res);
```
````
