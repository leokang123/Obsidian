const list = dv.pages('"Studied"')
  .where(p => p.tags && p.tags.includes("일반"))
  .where(p => !(p.tags ?? []).includes("대학교"));

const grouped = new Map();

for (const page of list) {
  const subject = page.주제 ?? "기타";
  if (!grouped.has(subject)) grouped.set(subject, []);
  grouped.get(subject).push(page);
}

const sortedSubjects = [...grouped.entries()].sort((a, b) => {
  const latestA = a[1]
    .map(p => window.moment(p.updated ?? p.file.mtime).valueOf())
    .sort((x, y) => y - x)[0] ?? 0;
  const latestB = b[1]
    .map(p => window.moment(p.updated ?? p.file.mtime).valueOf())
    .sort((x, y) => y - x)[0] ?? 0;
  return latestB - latestA;
});

if (!document.getElementById("study-link-style")) {
