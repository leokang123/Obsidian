```dataviewjs
const semester = "3-2";
const univTable = dv.pages("#대학교");
const filterSemester = univTable.filter(p => p.tags.includes(semester));
const univTag = filterSemester.tags.filter(p => p != "대학교" && p != semester);
const univSet = new Set(univTag);
const univList = [...univSet];
let res = '';
univList.forEach((tag) => {
	res += `~~~ dataview 
TABLE WITHOUT ID 
link(file.path,file.name) AS "${tag} :LiBookCheck: "
FROM #${tag} 
WHERE contains(file.tags, "대학교") and contains(file.tags, "${semester}")
SORT series DESC
~~~\n<br>\n\r`;
})
dv.paragraph(res);
```
