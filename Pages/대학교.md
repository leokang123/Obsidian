# 대학교 정리

````ad-orange
title: 4학년 2학기
```dataviewjs
const semester = "3-2"
const folderPath = "Legacy/대학교"
const list = dv.pages("#대학교").filter(p => p.file.folder === folderPath).filter(q => q.tags.includes(semester));
const semesters = ["3-1","3-2","4-1","4-2"]
const filterItem = ["대학교", "중요", "잡", "학습정리", "네이버", "개념", "프로젝트", semester]; // 세부적인 주제에 관한것만 
const sortPage = list.sort(page => page.수정일, "desc")
const tagSet = new Set(sortPage.tags);
const tagList = [...tagSet];
const filterList = tagList.filter((item) => !filterItem.includes(item)) // 세부적인 학습에 관련된거 필터링
let res = '';
filterList.forEach((tag) => {
res += `~~~ dataview
TABLE WITHOUT ID
link(file.path,file.name) AS "${tag} :LiBookCheck:"
FROM "${folderPath}" and #${tag}
WHERE !contains(file.tags, "일반")
SORT 주제, series DESC
~~~\n<br>\n\r`;
})
dv.paragraph(res);
```
````
