


### 최근이용 DB
```
TABLE WITHOUT ID
link(file.path,file.name) AS " 최근이용 :LiTimer:",  
dateformat(file.mtime, "yyyy년 MM월 dd일 HH:mm") AS ""
FROM "Studied" or "Tasks"
SORT file.mtime DESC 
LIMIT 10
```


### 일반 태그 DB
```
const list = dv.pages('#일반');
// const semesters = ["3-1","3-2","4-1","4-2"]
const filterItem = ["일반", "중요", "잡", "학습정리", "회고", "네이버", "개념" ,"대학교"];
const tagSet = new Set(list.tags);
const tagList = [...tagSet];

const filterList = tagList.filter((item) => !filterItem.includes(item))
let res = '';
filterList.forEach((tag) => {
res += `~~~ dataview 
TABLE WITHOUT ID
link(file.path,file.name) AS "${tag} :LiBookCheck: "
FROM "Studied" and #${tag} 
WHERE !contains(file.name, "대학교")
SORT series DESC
~~~\n<br>\n\r`;
})
dv.paragraph(res);
```

### 일반 태그 DB(개선판, 최근사용한거 위에 오게)
```
const list = dv.pages('#일반').filter(p => p.file.folder === "Studied");
// const semesters = ["3-1","3-2","4-1","4-2"]
const filterItem1 = ["대학교", "회고"] //개인 학습과 거리가 먼거 
const filterItem2 = ["일반", "중요", "잡", "학습정리", "회고", "네이버", "개념" ,"대학교"]; // 세부적인 주제에 관한것만 
const filterPage = list.filter(p => !p.tags.some(q => filterItem1.includes(q))); // 학습과 관련된거 필터링
const sortPage = filterPage.sort(page => page.수정일, "desc")
const tagSet = new Set(sortPage.tags);
const tagList = [...tagSet];
const filterList = tagList.filter((item) => !filterItem2.includes(item)) // 세부적인 학습에 관련된거 필터링
let res = '';
filterList.forEach((tag) => {
res += `~~~ dataview
TABLE WITHOUT ID
link(file.path,file.name) AS "${tag} :LiBookCheck:"
FROM "Studied" and #${tag}
WHERE !contains(file.name, "대학교")
SORT 주제, series DESC
~~~\n<br>\n\r`;
})
dv.paragraph(res);
```

### 학기 태그 DB
```
const semester = "3-2";
const filterItem = ["대학교", semester];
const univList = dv.pages(`#${semester}`);
const univSet = new Set(univList.tags);
const univTagList = [...univSet].filter((item) => !filterItem.includes(item));
let res = '';
univTagList.forEach((tag) => {
	if (tag === semester) return;
	res += `~~~ dataview 
TABLE WITHOUT ID 
link(file.path,file.name) AS "${tag} :LiBookCheck: "
FROM #${tag} 
WHERE contains(file.tags, "${semester}")
SORT series DESC
~~~\n<br>\n\r`;
})
dv.paragraph(res);
```
